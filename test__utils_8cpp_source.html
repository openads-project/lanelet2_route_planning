45TEST_F(ExtractRegulatoryElementsTest, AssignsRightOfWayRuleOnlyToYieldLanelet) { // NOLINT
46 auto priority_lanelet = makeLanelet(100, 0.0);
47 auto yield_lanelet = makeLanelet(200, -2.0);
48 auto stop_line = makeLine(300, 8.0, -3.0, 8.0, 1.0);
49 auto right_of_way = lanelet::RightOfWay::make(400, lanelet::AttributeMap(), {priority_lanelet}, {yield_lanelet}, stop_line);
50 priority_lanelet.addRegulatoryElement(right_of_way);
51 yield_lanelet.addRegulatoryElement(right_of_way);
52
53 const auto result = extractRegulatoryElements(priority_lanelet, {}, {yield_lanelet}, makePointSequence(0.0));
54
55 ASSERT_EQ(result.regulatory_element_msgs.size(), 1U);
56 EXPECT_TRUE(result.regulatory_element_idcs.empty());
57 ASSERT_EQ(result.adjacent_right_regulatory_element_idcs.size(), 1U);
58 EXPECT_EQ(result.adjacent_right_regulatory_element_idcs.front(), std::vector<uint8_t>({0U}));
59 EXPECT_EQ(result.regulatory_element_msgs.front().type, route_planning_msgs::msg::RegulatoryElement::TYPE_YIELD);
60}
