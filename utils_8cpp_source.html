37std::optional<lanelet::routing::Route> getRoute(const lanelet::routing::RoutingGraphUPtr& routing_graph,
38 const std::vector<lanelet::ConstLanelet>& route_lanelets) {
39 if (route_lanelets.empty()) {
40 return std::nullopt;
41 }
42
43 const lanelet::ConstLanelet& start_lanelet = route_lanelets.front();
44 std::vector<lanelet::ConstLanelet> intermediate_lanelets(route_lanelets.begin() + 1, route_lanelets.end() - 1);
45 const lanelet::ConstLanelet& destination_lanelet = route_lanelets.back();
46
47 // compute default route
48 const int routing_cost_id = 0; // RoutingCostDistance
49 const bool with_lane_changes = true;
50 auto route =
51 routing_graph->getRouteVia(start_lanelet, intermediate_lanelets, destination_lanelet, routing_cost_id, with_lane_changes);
52
53 // compute route alternatives with inverted start/destination lanelets (useful, if they are bidirectional)
54 auto route_alternative1 =
55 routing_graph->getRouteVia(start_lanelet.invert(), intermediate_lanelets, destination_lanelet, routing_cost_id);
56 auto route_alternative2 =
57 routing_graph->getRouteVia(start_lanelet, intermediate_lanelets, destination_lanelet.invert(), routing_cost_id);
58 auto route_alternative3 =
59 routing_graph->getRouteVia(start_lanelet.invert(), intermediate_lanelets, destination_lanelet.invert(), routing_cost_id);
60 std::vector<lanelet::routing::Route> route_alternatives;
61 if (route) route_alternatives.push_back(std::move(*route));
62 if (route_alternative1) route_alternatives.push_back(std::move(*route_alternative1));
63 if (route_alternative2) route_alternatives.push_back(std::move(*route_alternative2));
64 if (route_alternative3) route_alternatives.push_back(std::move(*route_alternative3));
65
66 // select shortest route
67 auto shortest_route_ptr = std::min_element(
68 route_alternatives.begin(), route_alternatives.end(),
69 [](const lanelet::routing::Route& a, const lanelet::routing::Route& b) { return a.length2d() < b.length2d(); });
70
71 if (shortest_route_ptr == route_alternatives.end()) {
72 return std::nullopt;
73 }
74 auto shortest_route = std::optional<lanelet::routing::Route>(std::move(*shortest_route_ptr));
75 return shortest_route;
76}

78size_t indexOfLineStringPointClosestToPoint(const std::vector<Eigen::Vector2d>& line_string,
79 const Eigen::Vector2d& point,
80 const bool consider_order,
81 const bool behind) {
82 if (line_string.empty()) {
83 return 0;
84 }
85
86 // loop over all points in line string to find closest one to given point
87 size_t idx_closest = 0;
88 double min_distance = std::numeric_limits<double>::infinity();
89 for (size_t i = 0; i < line_string.size(); ++i) {
90 double distance = (line_string[i] - point).norm();
91 if (distance < min_distance) {
92 min_distance = distance;
93 idx_closest = i;
94 }
95 }
96
97 // if considering order, make sure to return the point behind or ahead of the given point
98 if (consider_order) {
99 idx_closest = considerOrderForPointMatchedToLineString(line_string, point, idx_closest, behind);
100 }
101
102 return idx_closest;
103}

105size_t matchPointToLineString(const std::vector<Eigen::Vector2d>& line_string,
106 const Eigen::Vector2d& point,
107 const size_t idx_indication,
108 const bool consider_order,
109 const bool behind) {
110 if (line_string.empty()) {
111 return 0;
112 }
113
114 // constants
115 const double max_delta_s = 10.0;
116 const double max_local_distance = 10.0;
117
118 size_t idx_closest = 0;
119 double min_distance = std::numeric_limits<double>::infinity();
120 const size_t start_idx = std::min(idx_indication, line_string.size() - 1);
121
122 // loop over points in front of and behind the given index
123 for (int direction : {1, -1}) {
124 double delta_s = 0.0;
125 size_t idx = start_idx;
126 // only check points within the local range of max_delta_s
127 while (delta_s <= max_delta_s && idx < line_string.size()) {
128 double distance = (line_string[idx] - point).norm();
129 if (distance < min_distance) {
130 min_distance = distance;
131 idx_closest = idx;
132 }
133 if (direction == 1) {
134 if (idx + 1 >= line_string.size()) break;
135 delta_s += (line_string[idx + 1] - line_string[idx]).norm();
136 } else {
137 if (idx == 0) break;
138 delta_s += (line_string[idx] - line_string[idx - 1]).norm();
139 }
140 idx += direction;
141 }
142 }
143
144 // check if closest local point is within max distance, else find globally closest point
145 if (min_distance > max_local_distance) {
146 idx_closest = indexOfLineStringPointClosestToPoint(line_string, point, consider_order, behind);
147 } else if (consider_order) {
148 // if considering order, make sure to return the point behind or ahead of the given point
149 idx_closest = considerOrderForPointMatchedToLineString(line_string, point, idx_closest, behind);
150 }
151
152 return idx_closest;
153}

155size_t considerOrderForPointMatchedToLineString(const std::vector<Eigen::Vector2d>& line_string,
156 const Eigen::Vector2d& point,
157 const size_t idx_closest,
158 const bool behind) {
159 if (line_string.empty()) {
160 return 0;
161 }
162
163 size_t new_idx_closest = idx_closest;
164 Eigen::Vector2d closest_point_to_next;
165 const Eigen::Vector2d closest_point_to_point = point - line_string[idx_closest];
166 if (idx_closest + 1 < line_string.size()) {
167 closest_point_to_next = line_string[idx_closest + 1] - line_string[idx_closest];
168 } else if (idx_closest > 0) {
169 closest_point_to_next = line_string[idx_closest] - line_string[idx_closest - 1];
170 } else {
171 return idx_closest;
172 }
173
174 // use angle to check if closest point is behind or ahead of the given point
175 const double angle = angleBetweenVectors(closest_point_to_point, closest_point_to_next);
176 if (behind && std::abs(angle) > M_PI_2) {
177 new_idx_closest = idx_closest > 0 ? idx_closest - 1 : 0;
178 } else if (!behind && std::abs(angle) < M_PI_2) {
179 new_idx_closest = idx_closest + 1;
180 } else {
181 new_idx_closest = idx_closest;
182 }
183 new_idx_closest = std::clamp(new_idx_closest, size_t{0}, line_string.size() - 1);
184
185 return new_idx_closest;
186}

195std::vector<lanelet::ConstLanelet> adjacentLeftOrRightLanelets(const lanelet::ConstLanelet& lanelet,
196 const lanelet::routing::RoutingGraphUPtr& routing_graph,
197 bool left,
198 bool sort_from_left) {
199 std::vector<lanelet::ConstLanelet> adjacent_lanelets;
200 const int routing_cost_id = 0; // RoutingCostDistance
201 lanelet::routing::LaneletRelations relations =
202 left ? routing_graph->leftRelations(lanelet, routing_cost_id) : routing_graph->rightRelations(lanelet, routing_cost_id);
203 for (const auto& relation : relations) {
204 if ((left && (relation.relationType == lanelet::routing::RelationType::Left ||
205 relation.relationType == lanelet::routing::RelationType::AdjacentLeft)) ||
206 (!left && (relation.relationType == lanelet::routing::RelationType::Right ||
207 relation.relationType == lanelet::routing::RelationType::AdjacentRight))) {
208 adjacent_lanelets.push_back(relation.lanelet);
209 }
210 }
211
212 if ((left && sort_from_left) || (!left && !sort_from_left)) {
213 std::reverse(adjacent_lanelets.begin(), adjacent_lanelets.end());
214 }
215
216 return adjacent_lanelets;
217}

219std::vector<ProjectedLaneletPoints> projectPointToLaneletLines(const Eigen::Vector2d& point,
220 const Eigen::Vector2d& prev_point,
221 const Eigen::Vector2d& next_point,
222 const std::vector<lanelet::ConstLanelet>& lanelets,
223 const rclcpp::Logger& logger) {
224 std::vector<ProjectedLaneletPoints> projected_points_per_lanelet;
225
226 // loop over lanelets
227 for (const auto& lanelet : lanelets) {
228 ProjectedLaneletPoints projected_points;
229
230 // project point to left bounds
231 if (auto result = projectPointToLineStringAlongNormal(point, prev_point, next_point,
232 toEigen(lanelet.leftBound2d().basicLineString()))) {
233 projected_points.left_bound_point = result->projected_point;
234 } else {
235 RCLCPP_WARN(logger, "Failed to project point (%.3f, %.3f) to left bounds of lanelet %ld", point.x(), point.y(),
236 lanelet.id());
237 }
238
239 // project point to centerline
240 if (auto result = projectPointToLineStringAlongNormal(point, prev_point, next_point,
241 toEigen(lanelet.centerline2d().basicLineString()))) {
242 projected_points.centerline_point = result->projected_point;
243 } else {
244 RCLCPP_WARN(logger, "Failed to project point (%.3f, %.3f) to centerline of lanelet %ld", point.x(), point.y(),
245 lanelet.id());
246 }
247
248 // project point to right bounds
249 if (auto result = projectPointToLineStringAlongNormal(point, prev_point, next_point,
250 toEigen(lanelet.rightBound2d().basicLineString()))) {
251 projected_points.right_bound_point = result->projected_point;
252 } else {
253 RCLCPP_WARN(logger, "Failed to project point (%.3f, %.3f) to right bounds of lanelet %ld", point.x(), point.y(),
254 lanelet.id());
255 }
256
257 projected_points_per_lanelet.push_back(projected_points);
258 }
259
260 return projected_points_per_lanelet;
261}

263std::optional<int> computeFollowingLaneIdxOffset(const lanelet::ConstLanelet& lanelet,
264 const lanelet::ConstLanelet& lanelet_of_next_point,
265 const lanelet::routing::RoutingGraphUPtr& routing_graph) {
266 int following_lane_idx_offset = 0;
267 if (lanelet_of_next_point.id() != lanelet.id()) {
268 // get adjacent lanelets of current lanelet
269 std::vector<lanelet::ConstLanelet> adjacent_left_lanelets = adjacentLeftOrRightLanelets(lanelet, routing_graph, true);
270 std::vector<lanelet::ConstLanelet> adjacent_right_lanelets = adjacentLeftOrRightLanelets(lanelet, routing_graph, false);
271 const int suggested_lane_idx = static_cast<int>(adjacent_left_lanelets.size());
272
273 // get adjacent lanelets of next lanelet (lanelet of next point)
274 std::vector<lanelet::ConstLanelet> adjacent_left_lanelets_of_next_lanelet =
275 adjacentLeftOrRightLanelets(lanelet_of_next_point, routing_graph, true);
276 std::vector<lanelet::ConstLanelet> adjacent_right_lanelets_of_next_lanelet =
277 adjacentLeftOrRightLanelets(lanelet_of_next_point, routing_graph, false);
278 std::vector<lanelet::ConstLanelet> adjacent_lanelets_of_next_lanelet = adjacent_left_lanelets_of_next_lanelet;
279 adjacent_lanelets_of_next_lanelet.push_back(lanelet_of_next_point);
280 adjacent_lanelets_of_next_lanelet.insert(adjacent_lanelets_of_next_lanelet.end(),
281 adjacent_right_lanelets_of_next_lanelet.begin(),
282 adjacent_right_lanelets_of_next_lanelet.end());
283
284 // find following lanelet of current lanelet and adjacent lanelets in adjacent lanelets of next lanelet
285 std::vector<std::vector<lanelet::ConstLanelet>> lanelet_groups = {{lanelet}, adjacent_left_lanelets, adjacent_right_lanelets};
286 std::vector<int> lanelet_group_offset_factors = {0, 1, -1};
287 following_lane_idx_offset = std::numeric_limits<int>::max();
288
289 // first try to match following lanelet of current lanelet, then of adjacent left lanelets, then of adjacent right lanelets
290 for (size_t group_idx = 0; group_idx < lanelet_groups.size(); ++group_idx) {
291 const auto& lanelet_group = lanelet_groups[group_idx];
292 int group_offset_factor = lanelet_group_offset_factors[group_idx];
293
294 // loop over all lanelets in group
295 for (size_t a = 0; a < lanelet_group.size(); ++a) {
296 auto following_lanelets = routing_graph->following(lanelet_group[a], false);
297 if (following_lanelets.empty()) {
298 continue;
299 }
300 auto following_lanelet = following_lanelets.front();
301 size_t following_lanelet_idx = 0;
302 size_t follow_further_idx =
303 0; // counter for following lanelets more than once (e.g., if sampling skipped short lanelets)
304 size_t max_follow_further_iterations = 3; // maximum number of following lanelets to check
305
306 // loop until following lanelet is found in adjacent lanelets of next lanelet
307 while (following_lane_idx_offset == std::numeric_limits<int>::max()) {
308 // check all adjacent lanelets of next lanelet
309 for (size_t l = 0; l < adjacent_lanelets_of_next_lanelet.size(); ++l) {
310 if (following_lanelet.id() == adjacent_lanelets_of_next_lanelet[l].id()) {
311 following_lane_idx_offset =
312 static_cast<int>(l) - suggested_lane_idx + group_offset_factor * static_cast<int>(a + 1);
313 break;
314 }
315 }
316
317 // check abort conditions
318 follow_further_idx++;
319 if (follow_further_idx >= max_follow_further_iterations) {
320 break;
321 }
322
323 // get next following lanelet
324 following_lanelet_idx++;
325 if (following_lanelet_idx < following_lanelets.size()) {
326 following_lanelet = following_lanelets[following_lanelet_idx];
327 } else {
328 following_lanelets = routing_graph->following(following_lanelet, false);
329 if (following_lanelets.empty()) {
330 continue;
331 }
332 following_lanelet = following_lanelets.front();
333 following_lanelet_idx = 0;
334 }
335 }
336
337 if (following_lane_idx_offset != std::numeric_limits<int>::max()) {
338 break;
339 }
340 }
341
342 if (following_lane_idx_offset != std::numeric_limits<int>::max()) {
343 break;
344 }
345 }
346
347 if (following_lane_idx_offset == std::numeric_limits<int>::max()) {
348 // could not match following lanelets to adjacent lanelets of lanelet of next point
349 return std::nullopt;
350 }
351 }
352
353 return following_lane_idx_offset;
354}

386std::pair<Eigen::Vector2d, Eigen::Vector2d> extractDrivableSpace(const lanelet::LineStringLayer& line_string_layer,
387 const PointSequence& point_sequence,
388 const double max_distance) {
389 // find all line strings within max_distance
390 const lanelet::BasicLineString2d line_to_search_around = {point_sequence.current, point_sequence.next};
391 std::vector<std::pair<double, lanelet::ConstLineString3d>> line_strings_and_distances =
392 lanelet::geometry::findWithin2d(line_string_layer, line_to_search_around, max_distance);
393
394 // collect all intersections of the normal through the current point with nearby line strings
395 const std::vector<Eigen::Vector2d> normal_line = {
396 point_sequence.current,
397 point_sequence.current + normalOfPointAlongLineString(point_sequence.current, point_sequence.prev, point_sequence.next)};
398 std::vector<std::pair<Eigen::Vector2d, size_t>> projected_points_and_line_string_idcs;
399 for (size_t l = 0; l < line_strings_and_distances.size(); ++l) {
400 const auto& line_string = line_strings_and_distances[l].second;
401 const std::vector<Eigen::Vector2d> line_string_2d = to2d(toEigen(line_string.basicLineString()));
402 for (size_t i = 1; i < line_string_2d.size(); ++i) {
403 const std::vector<Eigen::Vector2d> line_segment = {line_string_2d[i - 1], line_string_2d[i]};
404 auto intersection = intersectionOfLines(normal_line, line_segment);
405 if (!intersection || !intersection->intersects_line2) {
406 continue;
407 }
408
409 const Eigen::Vector2d& projected_point = intersection->intersection;
410 if ((projected_point - point_sequence.current).norm() > max_distance) {
411 continue;
412 }
413
414 projected_points_and_line_string_idcs.emplace_back(projected_point, l);
415 }
416 }
417
418 // sort projected points by distance to current point
419 std::sort(projected_points_and_line_string_idcs.begin(), projected_points_and_line_string_idcs.end(),
420 [&point_sequence](const auto& a, const auto& b) {
421 return (a.first - point_sequence.current).norm() < (b.first - point_sequence.current).norm();
422 });
423
424 // split projected points into those left and those right of current point
425 const Eigen::Vector2d tangent = tangentOfPointAlongLineString(point_sequence.current, point_sequence.prev, point_sequence.next);
426 std::vector<std::pair<Eigen::Vector2d, size_t>> left_projected_points_and_line_string_idcs,
427 right_projected_points_and_line_string_idcs;
428 for (const auto& projected_point_and_line_string_idx : projected_points_and_line_string_idcs) {
429 // determine left/right based on angle between tangent and vector to projected point
430 const Eigen::Vector2d point_to_projected_point = projected_point_and_line_string_idx.first - point_sequence.current;
431 const double angle = angleBetweenVectors(tangent, point_to_projected_point);
432 if (angle > 0) {
433 left_projected_points_and_line_string_idcs.emplace_back(projected_point_and_line_string_idx);
434 } else {
435 right_projected_points_and_line_string_idcs.emplace_back(projected_point_and_line_string_idx);
436 }
437 }
438
439 // find drivable space bounds by following projected points until corresponding line string is not passable anymore
440 Eigen::Vector2d drivable_space_left, drivable_space_right;
441 bool is_drivable_space_left_limited_by_line_strings = false;
442 bool is_drivable_space_right_limited_by_line_strings = false;
443 for (const auto& projected_point_and_line_string_idx : left_projected_points_and_line_string_idcs) {
444 const auto& line_string = line_strings_and_distances[projected_point_and_line_string_idx.second].second;
445 if (!isLineStringDrivable(line_string)) {
446 drivable_space_left = projected_point_and_line_string_idx.first;
447 is_drivable_space_left_limited_by_line_strings = true;
448 break;
449 }
450 }
451 for (const auto& projected_point_and_line_string_idx : right_projected_points_and_line_string_idcs) {
452 const auto& line_string = line_strings_and_distances[projected_point_and_line_string_idx.second].second;
453 if (!isLineStringDrivable(line_string)) {
454 drivable_space_right = projected_point_and_line_string_idx.first;
455 is_drivable_space_right_limited_by_line_strings = true;
456 break;
457 }
458 }
459
460 // if drivable space is not limited by line strings, use maximum distance
461 const Eigen::Vector2d normal = normalOfPointAlongLineString(point_sequence.current, point_sequence.prev, point_sequence.next);
462 if (!is_drivable_space_left_limited_by_line_strings) {
463 drivable_space_left = point_sequence.current - normal * max_distance;
464 }
465 if (!is_drivable_space_right_limited_by_line_strings) {
466 drivable_space_right = point_sequence.current + normal * max_distance;
467 }
468
469 return {drivable_space_left, drivable_space_right};
470}

472bool isLineStringDrivable(const lanelet::ConstLineString3d& line_string) {
473 const std::unordered_set<std::string> drivable_types = {"arrow", "bike_marking", "centerline", "curbstone",
474 "lane_center", "line_thick", "line_thin", "pedestrian_marking",
475 "roadpainting", "stop_line", "traffic_light", "virtual",
476 "zebra_marking"};
477 if (line_string.hasAttribute("type")) {
478 std::string type = line_string.attribute("type").value();
479 if (drivable_types.count(type) > 0) {
480 if (type == "curbstone") {
481 if (!line_string.hasAttribute("subtype") || line_string.attribute("subtype").value() != "low") {
482 return false;
483 }
484 }
485 return true;
486 }
487 return line_string.hasAttribute("HoldingLine");
488 }
489 return false;
490}

492std::optional<RegulatoryElementCandidate> regulatoryElementCandidate(
493 const lanelet::ConstLanelet& lanelet, const std::shared_ptr<const lanelet::RegulatoryElement>& regulatory_element) {
494 // only consider right-of-way rules for lanelets that have to yield
495 const auto right_of_way = std::dynamic_pointer_cast<const lanelet::RightOfWay>(regulatory_element);
496 if (right_of_way && right_of_way->getManeuver(lanelet) != lanelet::ManeuverType::Yield) {
497 return std::nullopt;
498 }
499 const auto all_way_stop = std::dynamic_pointer_cast<const lanelet::AllWayStop>(regulatory_element);
500 const auto traffic_light = std::dynamic_pointer_cast<const lanelet::TrafficLight>(regulatory_element);
501
502 // create RegulatoryElement and skip unsupported types
503 RegulatoryElementCandidate candidate;
504 candidate.id = regulatory_element->id();
505 candidate.message.has_validity_stamp = false;
506 candidate.message.validity_stamp = builtin_interfaces::msg::Time();
507 std::tie(candidate.message.type, candidate.message.meta_value) = regulatoryElementType(regulatory_element);
508 if (candidate.message.type != route_planning_msgs::msg::RegulatoryElement::TYPE_YIELD &&
509 candidate.message.type != route_planning_msgs::msg::RegulatoryElement::TYPE_STOP &&
510 candidate.message.type != route_planning_msgs::msg::RegulatoryElement::TYPE_TRAFFIC_LIGHT &&
511 candidate.message.type != route_planning_msgs::msg::RegulatoryElement::TYPE_SPEED_LIMIT) {
512 return std::nullopt;
513 }
514
515 // extract reference line, using the lanelet-specific stop line for all-way stops
516 auto reference_line = regulatoryElementReferenceLine(regulatory_element);
517 bool has_stop_line = false;
518 if (all_way_stop) {
519 const auto participants = all_way_stop->lanelets();
520 if (std::find(participants.begin(), participants.end(), lanelet) == participants.end()) {
521 return std::nullopt;
522 }
523 candidate.lanelet_specific_reference_line = true;
524 reference_line = std::nullopt;
525 if (auto stop_line = all_way_stop->getStopLine(lanelet)) {
526 has_stop_line = true;
527 if (stop_line->size() < 2) {
528 return std::nullopt;
529 }
530 reference_line =
531 std::array<geometry_msgs::msg::Point, 2>{toRos(stop_line->front().basicPoint()), toRos(stop_line->back().basicPoint())};
532 }
533 } else if (right_of_way) {
534 has_stop_line = static_cast<bool>(right_of_way->stopLine());
535 } else if (traffic_light) {
536 has_stop_line = static_cast<bool>(traffic_light->stopLine());
537 }
538 if (has_stop_line && !reference_line) {
539 return std::nullopt;
540 }
541
542 // use lanelet end as reference line for yield, stop, and traffic-light rules without stop line
543 if (!has_stop_line && (right_of_way || all_way_stop || traffic_light)) {
544 if (lanelet.leftBound().empty() || lanelet.rightBound().empty()) {
545 return std::nullopt;
546 }
547 reference_line = std::array<geometry_msgs::msg::Point, 2>{toRos(lanelet.leftBound().back().basicPoint()),
548 toRos(lanelet.rightBound().back().basicPoint())};
549 candidate.lanelet_specific_reference_line = true;
550 }
551 if (!reference_line) {
552 return std::nullopt;
553 }
554
555 // set reference line and sign positions
556 candidate.message.reference_line = *reference_line;
557 candidate.message.positions = regulatoryElementPositions(regulatory_element);
558 return candidate;
559}

561bool keepYieldForRoute(const lanelet::routing::LaneletPath& path,
562 size_t yield_lanelet_idx,
563 const lanelet::RightOfWay& right_of_way,
564 const std::vector<const ParticipantRoutingGraph*>& priority_routing_graphs) {
565 // lanelets belong to the intersection if their interiors overlap with the intersection area
566 const auto overlapsIntersection = [](const lanelet::ConstLanelet& lanelet, const lanelet::BasicPolygonWithHoles2d& polygon) {
567 using Mask = boost::geometry::de9im::static_mask<'T', '*', '*', '*', '*', '*', '*', '*', '*'>;
568 return boost::geometry::relate(lanelet::CompoundHybridPolygon2d(lanelet.polygon2d()), polygon, Mask());
569 };
570
571 // keep yield if inputs are incomplete
572 if (priority_routing_graphs.empty() || yield_lanelet_idx >= path.size() || path.size() - yield_lanelet_idx < 2 ||
573 std::any_of(priority_routing_graphs.begin(), priority_routing_graphs.end(), [](const auto* participant_graph) {
574 return !participant_graph || !participant_graph->traffic_rules || !participant_graph->graph;
575 })) {
576 return true;
577 }
578
579 // get valid intersection area that overlaps with the route after the yielding lanelet
580 const auto intersection_areas = right_of_way.getParameters<lanelet::ConstArea>("intersection_area");
581 if (intersection_areas.size() != 1) {
582 return true;
583 }
584 const auto& intersection_area = intersection_areas.front();
585 if (!intersection_area.hasAttribute("subtype") || intersection_area.attribute("subtype").value() != "intersection") {
586 return true;
587 }
588 auto intersection_polygon = intersection_area.basicPolygonWithHoles2d();
589 boost::geometry::correct(intersection_polygon);
590 if (!boost::geometry::is_valid(intersection_polygon) ||
591 !overlapsIntersection(path[yield_lanelet_idx + 1], intersection_polygon)) {
592 return true;
593 }
594
595 // collect route lanelets through intersection, including approach and first exit to detect merges at its boundary
596 constexpr size_t max_intersection_lanelets = 256;
597 std::vector<lanelet::ConstLanelet> route_through_intersection{path[yield_lanelet_idx]};
598 size_t route_idx = yield_lanelet_idx + 1;
599 while (route_idx < path.size() && overlapsIntersection(path[route_idx], intersection_polygon)) {
600 route_through_intersection.push_back(path[route_idx]);
601 if (route_through_intersection.size() > max_intersection_lanelets + 1) {
602 return true;
603 }
604 ++route_idx;
605 }
606 if (route_idx == path.size()) {
607 return true;
608 }
609 route_through_intersection.push_back(path[route_idx]);
610
611 // get priority lanelets
612 const auto priority_lanelets = right_of_way.rightOfWayLanelets();
613 if (priority_lanelets.empty()) {
614 return true;
615 }
616
617 // check whether a priority lanelet shares a lanelet or 2D centerline point with the route through the intersection
618 constexpr size_t max_conflict_checks = 8192;
619 size_t conflict_checks = 0;
620 const auto conflicts_with_route_or_reaches_limit = [&](const lanelet::ConstLanelet& priority_lanelet) {
621 const auto priority_centerline = lanelet::traits::toHybrid(priority_lanelet.centerline2d());
622 return std::any_of(route_through_intersection.begin(), route_through_intersection.end(), [&](const auto& route_lanelet) {
623 return ++conflict_checks > max_conflict_checks || priority_lanelet.id() == route_lanelet.id() ||
624 lanelet::geometry::intersects(priority_centerline, lanelet::traits::toHybrid(route_lanelet.centerline2d()));
625 });
626 };
627
628 // loop over priority lanelets, participants, and legal directions
629 for (const auto& priority_lanelet : priority_lanelets) {
630 bool traversed_priority = false;
631 for (const auto* participant_graph : priority_routing_graphs) {
632 for (const auto& approach : {priority_lanelet, priority_lanelet.invert()}) {
633 if (!participant_graph->traffic_rules->canPass(approach)) {
634 continue;
635 }
636
637 // skip directions leading away from the intersection
638 const bool approach_in_intersection = overlapsIntersection(approach, intersection_polygon);
639 const auto approach_successors = participant_graph->graph->following(approach, false);
640 if (!approach_in_intersection &&
641 std::none_of(approach_successors.begin(), approach_successors.end(),
642 [&](const auto& successor) { return overlapsIntersection(successor, intersection_polygon); })) {
643 continue;
644 }
645 traversed_priority = true;
646
647 // traverse all lanelets reachable inside the intersection and check them for conflicts
648 std::deque<lanelet::ConstLanelet> pending{approach};
649 std::set<std::pair<lanelet::Id, bool>> visited;
650 bool exited_intersection = false;
651 while (!pending.empty()) {
652 const auto current = pending.front();
653 pending.pop_front();
654 if (!visited.insert({current.id(), current.inverted()}).second) {
655 continue;
656 }
657 if (visited.size() > max_intersection_lanelets) {
658 return true;
659 }
660 if (conflicts_with_route_or_reaches_limit(current)) {
661 return true;
662 }
663
664 // keep yield for dead ends inside the intersection
665 const auto successors = participant_graph->graph->following(current, false);
666 if (successors.empty()) {
667 if (boost::geometry::covered_by(current.centerline2d().back().basicPoint(), intersection_polygon)) {
668 return true;
669 }
670 exited_intersection = true;
671 continue;
672 }
673
674 // continue with successors inside the intersection and check first exits
675 for (const auto& successor : successors) {
676 if (overlapsIntersection(successor, intersection_polygon)) {
677 pending.push_back(successor);
678 } else {
679 // keep yield if an approach branch bypasses the intersection
680 if (current == approach && !approach_in_intersection) {
681 return true;
682 }
683 exited_intersection = true;
684 if (conflicts_with_route_or_reaches_limit(successor)) {
685 return true;
686 }
687 }
688 }
689 }
690
691 // keep yield if the intersection is never exited
692 if (!exited_intersection) {
693 return true;
694 }
695 }
696 }
697
698 // keep yield if no participant enters the intersection from the priority lanelet
699 if (!traversed_priority) {
700 return true;
701 }
702 }
703 return false;
704}

706std::vector<std::vector<RegulatoryElementCandidate>> regulatoryElementsAlongRoute(
707 const lanelet::routing::LaneletPath& path,
708 const std::vector<Eigen::Vector2d>& reference_line,
709 const std::vector<size_t>& lanelet_idx_by_point,
710 const std::vector<const ParticipantRoutingGraph*>& priority_routing_graphs) {
711 // init result
712 std::vector<std::vector<RegulatoryElementCandidate>> result(reference_line.size());
713 if (reference_line.size() < 2 || reference_line.size() != lanelet_idx_by_point.size()) {
714 return result;
715 }
716
717 // loop over lanelets of path
718 for (size_t lanelet_idx = 0; lanelet_idx < path.size(); ++lanelet_idx) {
719 // find first route segment of lanelet, including the segment entering it
720 const auto first_point = std::lower_bound(lanelet_idx_by_point.begin(), lanelet_idx_by_point.end(), lanelet_idx);
721 if (first_point == lanelet_idx_by_point.end()) {
722 break;
723 }
724 const size_t first_point_idx = std::distance(lanelet_idx_by_point.begin(), first_point);
725 const size_t first_segment = first_point_idx > 0 ? first_point_idx - 1 : 0;
726
727 // loop over regulatory elements of lanelet
728 const auto& lanelet = path[lanelet_idx];
729 for (const auto& regulatory_element : lanelet.regulatoryElements()) {
730 auto candidate = regulatoryElementCandidate(lanelet, regulatory_element);
731 if (!candidate) {
732 continue;
733 }
734
735 // omit yield if no priority traffic conflicts with route
736 if (candidate->message.type == route_planning_msgs::msg::RegulatoryElement::TYPE_YIELD) {
737 const auto right_of_way = std::dynamic_pointer_cast<const lanelet::RightOfWay>(regulatory_element);
738 if (right_of_way && !keepYieldForRoute(path, lanelet_idx, *right_of_way, priority_routing_graphs)) {
739 continue;
740 }
741 }
742
743 // assign regulatory element to route element before first intersected route segment, also beyond lanelet end
744 const auto& line = candidate->message.reference_line;
745 const std::vector<Eigen::Vector2d> effect_line = {toEigen2d(line[0]), toEigen2d(line[1])};
746 for (size_t c = first_segment; c + 1 < reference_line.size(); ++c) {
747 const std::vector<Eigen::Vector2d> route_segment = {reference_line[c], reference_line[c + 1]};
748 const auto intersection = intersectionOfLines(effect_line, route_segment);
749 if (!intersection || !intersection->intersects_line1 || !intersection->intersects_line2) {
750 continue;
751 }
752 // check if regulatory element has already been assigned (by another lanelet)
753 auto& assigned = result[c];
754 const bool already_assigned = !candidate->lanelet_specific_reference_line &&
755 std::any_of(assigned.begin(), assigned.end(), [&](const auto& existing) {
756 return existing.id == candidate->id && !existing.lanelet_specific_reference_line;
757 });
758 if (!already_assigned) {
759 assigned.push_back(*candidate);
760 }
761 break;
762 }
763 }
764 }
765 return result;
766}

768ExtractRegulatoryElementsResult extractRegulatoryElements(const lanelet::ConstLanelet& lanelet,
769 const std::vector<lanelet::ConstLanelet>& adjacent_left_lanelets,
770 const std::vector<lanelet::ConstLanelet>& adjacent_right_lanelets,
771 const PointSequence& point_sequence,
772 bool include_main_lanelet) {
773 // init result
774 ExtractRegulatoryElementsResult result;
775 result.adjacent_left_regulatory_element_idcs.resize(adjacent_left_lanelets.size());
776 result.adjacent_right_regulatory_element_idcs.resize(adjacent_right_lanelets.size());
777
778 // gather lanelets in single vector (left adjacent, current, right adjacent)
779 std::vector<lanelet::ConstLanelet> lanelets = adjacent_left_lanelets;
780 lanelets.push_back(lanelet);
781 lanelets.insert(lanelets.end(), adjacent_right_lanelets.begin(), adjacent_right_lanelets.end());
782
783 // loop over lanelets
784 std::unordered_map<lanelet::Id, size_t> regulatory_element_msg_idx_by_id;
785 for (size_t l = 0; l < lanelets.size(); ++l) {
786 if (!include_main_lanelet && l == adjacent_left_lanelets.size()) {
787 continue;
788 }
789 const auto& current_lanelet = lanelets[l];
790
791 // project point sequence onto centerline of adjacent lanelets
792 PointSequence lanelet_point_sequence = point_sequence;
793 if (current_lanelet.id() != lanelet.id()) {
794 const auto centerline = toEigen(current_lanelet.centerline2d().basicLineString());
795 lanelet_point_sequence = {projectPointToLineString(point_sequence.prev, centerline),
796 projectPointToLineString(point_sequence.current, centerline),
797 projectPointToLineString(point_sequence.next, centerline)};
798 }
799
800 // loop over regulatory elements of lanelet
801 for (const auto& regulatory_element : current_lanelet.regulatoryElements()) {
802 auto candidate = regulatoryElementCandidate(current_lanelet, regulatory_element);
803 if (!candidate) {
804 continue;
805 }
806
807 // only consider regulatory element if reference line intersects with forward segment of point sequence
808 const auto& line = candidate->message.reference_line;
809 const std::vector<Eigen::Vector2d> effect_line = {toEigen2d(line[0]), toEigen2d(line[1])};
810 const std::vector<Eigen::Vector2d> route_segment = {lanelet_point_sequence.current, lanelet_point_sequence.next};
811 const auto intersection = intersectionOfLines(effect_line, route_segment);
812 if (!intersection || !intersection->intersects_line1 || !intersection->intersects_line2) {
813 continue;
814 }
815 // assign intersection at shared endpoint to preceding segment if it also intersects there
816 if (intersection->intersection.isApprox(lanelet_point_sequence.current)) {
817 const auto previous = intersectionOfLines(effect_line, {lanelet_point_sequence.prev, lanelet_point_sequence.current});
818 if (previous && previous->intersects_line1 && previous->intersects_line2) {
819 continue;
820 }
821 }
822
823 // check if regulatory element has already been extracted (by another lanelet)
824 size_t regulatory_element_msg_idx = 0;
825 if (!candidate->lanelet_specific_reference_line && regulatory_element_msg_idx_by_id.count(candidate->id) > 0) {
826 regulatory_element_msg_idx = regulatory_element_msg_idx_by_id[candidate->id];
827 } else {
828 // add regulatory element to result
829 regulatory_element_msg_idx = result.regulatory_element_msgs.size();
830 regulatory_element_msg_idx_by_id[candidate->id] = regulatory_element_msg_idx;
831 result.regulatory_element_msgs.push_back(candidate->message);
832 result.regulatory_element_ids.push_back(candidate->id);
833 }
834
835 // assign regulatory element to respective lanelet in result
836 if (l < adjacent_left_lanelets.size()) {
837 result.adjacent_left_regulatory_element_idcs[l].push_back(regulatory_element_msg_idx);
838 } else if (l == adjacent_left_lanelets.size()) {
839 result.regulatory_element_idcs.push_back(regulatory_element_msg_idx);
840 } else {
841 const size_t l_right = l - adjacent_left_lanelets.size() - 1;
842 result.adjacent_right_regulatory_element_idcs[l_right].push_back(regulatory_element_msg_idx);
843 }
844 }
845 }
846
847 return result;
848}

1026uint8_t speedLimit(const lanelet::ConstLanelet& lanelet, const Eigen::Vector2d& point) {
1027 uint8_t speed_limit = route_planning_msgs::msg::RegulatoryElement::META_VALUE_SPEED_UNLIMITED;
1028 std::vector<Eigen::Vector2d> centerline = toEigen(lanelet.centerline2d().basicLineString());
1029 double best_reference_arc_length = 0.0;
1030 bool found_valid_regulatory_element_speed_limit = false;
1031 bool found_future_reference_line = false;
1032 bool found_past_cancel_line = false;
1033
1034 // loop over regulatory elements of lanelet
1035 const auto regulatory_elements = lanelet.regulatoryElements();
1036 for (const auto& regulatory_element : regulatory_elements) {
1037 // extract type and meta value, skip non-speed-limit regulatory elements
1038 uint8_t regulatory_element_type = route_planning_msgs::msg::RegulatoryElement::TYPE_UNKNOWN;
1039 uint8_t regulatory_element_meta_value = route_planning_msgs::msg::RegulatoryElement::META_VALUE_UNKNOWN;
1040 std::tie(regulatory_element_type, regulatory_element_meta_value) = regulatoryElementType(regulatory_element);
1041 if (regulatory_element_type != route_planning_msgs::msg::RegulatoryElement::TYPE_SPEED_LIMIT) {
1042 continue;
1043 }
1044

1326 for (size_t r = 0; r < route_elements.size(); ++r) {
1327 // get current, previous and next route element

1330 const auto& next_route_element = (r < route_elements.size() - 1) ? route_elements[r + 1] : route_element;

1345 const auto next_lane_element_opt =

1351 const bool changes_lane_to_next_point = next_route_element.will_change_suggested_lane;
1352 const auto prev_point_for_orientation = (changes_lane_from_prev_point || !prev_lane_element_opt)

1354 : toEigen2d(prev_lane_element_opt->reference_pose.position);
1355 const auto next_point_for_orientation = (changes_lane_to_next_point || !next_lane_element_opt)
1356 ? point
1357 : toEigen2d(next_lane_element_opt->reference_pose.position);

1360 const auto orientation = tangentOfPointAlongLineString(point, prev_point_for_orientation, next_point_for_orientation);
1361 lane_element.reference_pose.orientation = toRosQuaternion(orientation);

1366 for (int r = static_cast<int>(route_elements.size()) - 1; r >= 0; --r) {

1380 auto* curr_lane_element = &lane_element;
1381 auto* prev_route_element = &route_elements[r - 1];
1382 auto prev_lane_element_idx_opt = route_planning_msgs::route_access::getPrecedingLaneElementIdx(l, *prev_route_element);

1422void sortLaneletsByMatchingCost(const lanelet::BasicPoint2d& point,
1423 std::vector<std::pair<double, lanelet::ConstLanelet>>& lanelets_with_distances,
1424 const std::optional<lanelet::traffic_rules::TrafficRulesPtr>& traffic_rules) {

1426 std::pair<double, lanelet::ConstLanelet> lanelet_with_distance;
1427 double cost;
1428 };
1429
1430 std::vector<LaneletMatchingCost> lanelet_matching_costs;
1431 lanelet_matching_costs.reserve(lanelets_with_distances.size());
1432
1433 // compute the matching cost for each candidate lanelet
1434 for (const auto& lanelet_with_distance : lanelets_with_distances) {
1435 const lanelet::ConstLanelet& lanelet = lanelet_with_distance.second;
1436 const lanelet::ArcCoordinates arc_coordinates = lanelet::geometry::toArcCoordinates(lanelet.centerline2d(), point);
1437 const bool point_is_inside_lanelet = lanelet::geometry::inside(lanelet, point);
1438
1439 // favor lanelets containing the point and penalize lateral distance quadratically
1440 double cost = (point_is_inside_lanelet ? 0.5 : 2.0) * arc_coordinates.distance * arc_coordinates.distance;
1441
1442 // keep non-passable lanelets as fallback candidates, but rank them lower
1443 if (traffic_rules && !traffic_rules.value()->canPass(lanelet)) {
1444 cost *= 2.0;
1445 }
1446
1447 lanelet_matching_costs.push_back({lanelet_with_distance, cost});
1448 }
1449
1450 // apply the sorted order to the original lanelet-distance pairs
1451 std::sort(lanelet_matching_costs.begin(), lanelet_matching_costs.end(),
1452 [](const LaneletMatchingCost& lhs, const LaneletMatchingCost& rhs) { return lhs.cost < rhs.cost; });
1453
1454 for (size_t i = 0; i < lanelet_matching_costs.size(); ++i) {
1455 lanelets_with_distances.at(i) = lanelet_matching_costs[i].lanelet_with_distance;
1456 }
1457}
